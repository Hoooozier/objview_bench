# install Enroot (https://github.com/NVIDIA/enroot/blob/master/doc/installation.md)
arch=$(dpkg --print-architecture)
curl -fSsL -O https://github.com/NVIDIA/enroot/releases/download/v4.1.2/enroot_4.1.2-1_${arch}.deb
sudo apt-get update
sudo apt install -y ./enroot_4.1.2-1_${arch}.deb
rm enroot_4.1.2-1_${arch}.deb

# Import Container
sudo enroot import docker://nvidia/cuda:11.8.0-devel-ubuntu20.04
enroot create --name ObjViewBench nvidia+cuda+11.8.0-devel-ubuntu20.04.sqsh
sudo rm nvidia+cuda+11.8.0-devel-ubuntu20.04.sqsh

# Install NVIDIA Container Toolkit (https://docs.nvidia.com/datacenter/cloud-native/container-toolkit/latest/install-guide.html)
curl -fsSL https://nvidia.github.io/libnvidia-container/gpgkey | sudo gpg --dearmor -o /usr/share/keyrings/nvidia-container-toolkit-keyring.gpg \
  && curl -s -L https://nvidia.github.io/libnvidia-container/stable/deb/nvidia-container-toolkit.list | \
    sed 's#deb https://#deb [signed-by=/usr/share/keyrings/nvidia-container-toolkit-keyring.gpg] https://#g' | \
    sudo tee /etc/apt/sources.list.d/nvidia-container-toolkit.list
sudo apt-get update
sudo apt-get install -y nvidia-container-toolkit

# (If needed) Export/Re-import Container
enroot export -o ObjViewBench.sqsh ObjViewBench
enroot create --name ObjViewBench ObjViewBench.sqsh

# (If needed) Remove Container
enroot remove ObjViewBench

# Use Enroot
enroot start -w -r -e DISPLAY=$DISPLAY -e NVIDIA_VISIBLE_DEVICES=all -m ~:/home/${USER} -m /mnt:/mnt -m /dev:/dev -m /tmp/.X11-unix:/tmp/.X11-unix ObjViewBench

# Install C++ Environment (opencv, pcl, eigen, octomap, gurobi)
apt-get update
apt-get install lsb-release git wget build-essential libssl-dev vim

# Install opencv 4.2.0 and pcl 1.10.0 using default in ros noetic (https://wiki.ros.org/noetic/Installation/Ubuntu)
sh -c 'echo "deb http://packages.ros.org/ros/ubuntu $(lsb_release -sc) main" > /etc/apt/sources.list.d/ros-latest.list'
apt install curl
curl -s https://raw.githubusercontent.com/ros/rosdistro/master/ros.asc | apt-key add -
apt update
apt install ros-noetic-desktop-full
apt-get clean

# Install cmake 3.24
wget https://github.com/Kitware/CMake/releases/download/v3.24.0/cmake-3.24.0.tar.gz
tar -zxvf cmake-3.24.0.tar.gz
cd cmake-3.24.0
./bootstrap
make -j8
make install
cd /
rm cmake-3.24.0.tar.gz

# Install eigen 3.4.0
wget https://gitlab.com/libeigen/eigen/-/archive/3.4.0/eigen-3.4.0.tar.gz
tar -xzf eigen-3.4.0.tar.gz
cd eigen-3.4.0
mkdir build
cd build
cmake ..
make install
cd /
rm eigen-3.4.0.tar.gz

# Install octomap 1.10.0
git clone https://github.com/OctoMap/octomap.git
cd octomap
git checkout v1.10.0
git describe --tags
mkdir build
cd build
cmake ..
make -j8
make install
cd /

# Install gurobi 13.0.1
wget https://packages.gurobi.com/13.0/gurobi13.0.1_linux64.tar.gz
tar -xzvf gurobi13.0.1_linux64.tar.gz
vim ~/.bashrc
## Add to .bashrc
export GUROBI_HOME="/gurobi1301/linux64"
export GRB_LICENSE_FILE="/home/${USER}/gurobi.lic"
export PATH="${PATH}:${GUROBI_HOME}/bin"
export LD_LIBRARY_PATH="${LD_LIBRARY_PATH}:${GUROBI_HOME}/lib"
source ~/.bashrc
cd gurobi1301/linux64/src/build
make -j8
cp libgurobi_c++.a ../../lib/
cd /
rm gurobi13.0.1_linux64.tar.gz
## Setting up the Free academic license file using https://www.gurobi.com/features/academic-named-user-license/ (if the MAC may change, use WLS Academic), make sure the install path is same as GRB_LICENSE_FILE
grbgetkey {YOUR_KEY} [Or download your server gurobi.lic and move to GRB_LICENSE_FILE]
